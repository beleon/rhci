## 1. readFile, writeFile, appendFile

-- Run in a fresh directory holding oracle/files_io (see oracle/README.md).
ghci> readFile "nofile"
*** Exception: nofile: openFile: does not exist (No such file or directory)
ghci> readFile ""
*** Exception: : openFile: does not exist (No such file or directory)
ghci> readFile "/"
*** Exception: /: openFile: inappropriate type (is a directory)
ghci> readFile "dir"
*** Exception: dir: openFile: inappropriate type (is a directory)
ghci> readFile "bad.txt/x"
*** Exception: bad.txt/x: openFile: inappropriate type (Not a directory)
ghci> readFile "text.txt"
"line one\n\955\8594x\nlast"
ghci> lines <$> readFile "text.txt"
["line one","\955\8594x","last"]
ghci> length <$> readFile "text.txt"
17
ghci> writeFile "a.txt" "hello\nworld\n"
ghci> readFile "a.txt"
"hello\nworld\n"
ghci> appendFile "a.txt" "more\n"
ghci> readFile "a.txt" >>= putStr
hello
world
more
ghci> appendFile "new.txt" "created"
ghci> readFile "new.txt"
"created"
ghci> writeFile "e" ""
ghci> readFile "e"
""
ghci> writeFile "n" "a\nb\r\nc"
ghci> lines <$> readFile "n"
["a","b\r","c"]
ghci> writeFile "u.txt" "λ→\955"
ghci> readFile "u.txt"
"\955\8594\955"
ghci> length <$> readFile "u.txt"
3
ghci> writeFile "nodir/x" "a"
*** Exception: nodir/x: withFile: does not exist (No such file or directory)
ghci> appendFile "nodir/x" "a"
*** Exception: nodir/x: withFile: does not exist (No such file or directory)
ghci> writeFile "dir" "x"
*** Exception: dir: withFile: inappropriate type (Is a directory)
ghci> appendFile "dir" "x"
*** Exception: dir: withFile: inappropriate type (Is a directory)

*** Exception: bad.txt/x: withFile: inappropriate type (Not a directory)

*** Exception: : withFile: does not exist (No such file or directory)

## 2. Lazy reading and locks

ghci> writeFile "a.txt" "0123456789"
ghci> s <- readFile "a.txt"
ghci> take 3 s
"012"
ghci> writeFile "a.txt" "x"
*** Exception: a.txt: withFile: resource busy (file is locked)
ghci> appendFile "./a.txt" "x"
*** Exception: ./a.txt: withFile: resource busy (file is locked)
ghci> length s
10
ghci> writeFile "a.txt" "x"
ghci> readFile "a.txt"
"x"
ghci> x <- readFile "a.txt"
ghci> y <- readFile "a.txt"
ghci> x
"x"
ghci> writeFile "a.txt" "q"
*** Exception: a.txt: withFile: resource busy (file is locked)
ghci> y
"x"
ghci> writeFile "a.txt" "q"
ghci> readFile "a.txt" >>= \s -> writeFile "a.txt" (map succ s)
*** Exception: a.txt: withFile: resource busy (file is locked)
ghci> readFile "a.txt" >>= \s -> length s `seq` writeFile "a.txt" (s ++ "!")
*** Exception: a.txt: withFile: resource busy (file is locked)
ghci> readFile "a.txt"
"q"
ghci> r <- readFile "bad2.txt"
ghci> length r
*** Exception: bad2.txt: hGetContents: invalid argument (cannot decode byte sequence starting from 255)
ghci> writeFile "bad2.txt" "ok"
ghci> readFile "bad2.txt"
"ok"
