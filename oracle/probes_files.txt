## 2. Lazy reading and locks
writeFile "a.txt" "0123456789"
s <- readFile "a.txt"
take 3 s
writeFile "a.txt" "x"
appendFile "./a.txt" "x"
length s
writeFile "a.txt" "x"
readFile "a.txt"
x <- readFile "a.txt"
y <- readFile "a.txt"
x
writeFile "a.txt" "q"
y
writeFile "a.txt" "q"
readFile "a.txt" >>= \s -> writeFile "a.txt" (map succ s)
readFile "a.txt" >>= \s -> length s `seq` writeFile "a.txt" (s ++ "!")
readFile "a.txt"
r <- readFile "bad2.txt"
length r
writeFile "bad2.txt" "ok"
readFile "bad2.txt"
